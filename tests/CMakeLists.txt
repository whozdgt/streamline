# Tests of Pods' sorting that need no Qt: cmake -S tests -B build-tests && cmake --build build-tests
#   podsort_test                         key words
#   nest_test spec_sample.txt            nested sorting and spec matching by key words
#   podai_test meaning.gguf [writer.gguf] [spec.txt]   the on-device AI (models from the ai-models release)
#   match_eval meaning.gguf spec.pdf match/notes.txt   how often documents go to the right spec point
cmake_minimum_required(VERSION 3.21)
project(pods_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
include(${CMAKE_CURRENT_LIST_DIR}/../cmake/llama.cmake)
set(SRC ${CMAKE_CURRENT_LIST_DIR}/../src)
add_executable(podsort_test podsort_test.cpp ${SRC}/podsort.cpp ${SRC}/specread.cpp)
target_include_directories(podsort_test PRIVATE ${SRC})
add_executable(nest_test nest_test.cpp ${SRC}/podsort.cpp ${SRC}/specread.cpp)
target_include_directories(nest_test PRIVATE ${SRC})
add_executable(podai_test podai_test.cpp ${SRC}/podai.cpp ${SRC}/podsort.cpp ${SRC}/specread.cpp)
target_include_directories(podai_test PRIVATE ${SRC})
target_link_libraries(podai_test PRIVATE llama)
add_executable(spec_test spec_test.cpp ${SRC}/podsort.cpp ${SRC}/specread.cpp ${SRC}/pdftext.cpp ${SRC}/pdfcrypto.cpp)
target_include_directories(spec_test PRIVATE ${SRC})
add_executable(pdftext_test pdftext_test.cpp ${SRC}/pdftext.cpp ${SRC}/pdfcrypto.cpp)
target_include_directories(pdftext_test PRIVATE ${SRC})
add_executable(match_eval match_eval.cpp ${SRC}/podai.cpp ${SRC}/podsort.cpp ${SRC}/specread.cpp ${SRC}/pdftext.cpp ${SRC}/pdfcrypto.cpp)
target_include_directories(match_eval PRIVATE ${SRC})
target_link_libraries(match_eval PRIVATE llama)
